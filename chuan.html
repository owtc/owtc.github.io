btnScanQR.addEventListener('click', async () => { if (!scannerAvailable) { addMsg('❌ 扫描库未加载，请检查网络或使用 HTTPS'); return; } // 先检测摄像头权限 try { const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: 'environment' } }); stream.getTracks().forEach(track => track.stop()); } catch (err) { if (err.name === 'NotAllowedError' || err.message.includes('Permission denied')) { addMsg('❌ 摄像头权限被拒绝，请在浏览器设置中允许访问摄像头'); } else { addMsg('❌ 无法访问摄像头: ' + err.message); } return; } if (qrScanner) { qrScanner.stop().then(() => { ... }).catch(()=>{}); } qrScanner = new Html5Qrcode("qrReader"); const config = { fps: 10, qrbox: { width: 250, height: 250 } }; Html5Qrcode.getCameras().then(devices => { if (devices && devices.length) { // 优先选择后置摄像头 let cameraId = devices[0].id; for (const device of devices) { const label = device.label.toLowerCase(); if (label.includes('back') || label.includes('environment')) { cameraId = device.id; break; } } qrScanner.start(cameraId, config, onSuccess, onError).catch(err => { // 启动扫描失败 if (err.name === 'NotAllowedError' || err.message.includes('Permission denied')) { addMsg('❌ 摄像头权限被拒绝，请在浏览器设置中允许访问摄像头'); } else { addMsg('❌ 无法启动摄像头: ' + err.message); } qrScanner = null; }); } else { addMsg('❌ 未找到可用摄像头'); qrScanner = null; } }).catch(err => { if (err.name === 'NotAllowedError' || err.message.includes('Permission denied')) { addMsg('❌ 摄像头权限被拒绝，请在浏览器设置中允许访问摄像头'); } else { addMsg('❌ 获取摄像头列表失败: ' + err.message); } qrScanner = null; }); });
